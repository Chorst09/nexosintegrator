# ============================================
# COMANDOS PARA EXECUTAR NO SERVIDOR
# Copie e cole linha por linha no terminal
# ============================================

# 1. INSTALAR DEPENDÊNCIAS
curl -fsSL https://deb.nodesource.com/setup_20.x | bash -
apt-get install -y nodejs postgresql postgresql-contrib nginx

# 2. CONFIGURAR POSTGRESQL
sudo -u postgres createuser nexoscrm
sudo -u postgres createdb nexoscrm
sudo -u postgres psql -c "ALTER USER nexoscrm PASSWORD '<DB_PASSWORD>';"
sudo -u postgres psql -c "GRANT ALL PRIVILEGES ON DATABASE nexoscrm TO nexoscrm;"

# 3. CRIAR DIRETÓRIO
mkdir -p /var/www/nexoscrm
cd /var/www/nexoscrm

# 4. BAIXAR E EXTRAIR ARQUIVOS
# (Execute o comando scp do seu computador primeiro)
tar -xzf nexoscrm-simple.tar.gz

# 5. CONFIGURAR BACKEND
cd backend
npm install --production
cp ../.env.production .env
npx prisma generate
npx prisma migrate deploy
node prisma/migrate-production.js

# 6. CONFIGURAR FRONTEND  
cd ../frontend
npm install
VITE_API_URL=http://209.50.241.25/api npm run build

# 7. CONFIGURAR NGINX
cat > /etc/nginx/sites-available/nexoscrm << 'EOF'
server {
    listen 80;
    server_name _;
    
    location / {
        root /var/www/nexoscrm/frontend/dist;
        try_files $uri $uri/ /index.html;
    }
    
    location /api/ {
        proxy_pass http://localhost:3001;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }
    
    location /uploads/ {
        proxy_pass http://localhost:3001;
    }
}
EOF

ln -s /etc/nginx/sites-available/nexoscrm /etc/nginx/sites-enabled/
rm -f /etc/nginx/sites-enabled/default
nginx -t && systemctl reload nginx

# 8. INSTALAR PM2 E INICIAR
npm install -g pm2
cd /var/www/nexoscrm/backend
pm2 start server.js --name nexoscrm-api
pm2 startup
pm2 save

# 9. VERIFICAR STATUS
pm2 status
systemctl status nginx
systemctl status postgresql

echo "✅ Deploy concluído!"
echo "🌐 Acesse: http://209.50.241.25"
echo "📧 Login: admin@nexoscrm.com"
echo "🔑 Senha: <ADMIN_PASSWORD>"
